# FlyArchOs — триггер облачной сборки ISO в GitHub Actions.
#
# Как запустить сборку:
#   1. укажите профиль ниже (slim или full);
#   2. сохраните файл и запушьте изменение — workflow «Build ISO» стартует сам
#      (push-триггер смотрит на этот файл).
#
#   slim — оба окружения (Hyprland + KDE Plasma) без дополнительных наборов,
#          ~3-4 ГБ, ~30-50 минут;
#   full — «жирный» образ со всеми наборами ПО, ~10 ГБ, до 5 часов
#          (может не влезть в место стандартного раннера).
#
# Готовый ISO — в артефактах запуска (хранится 7 дней).
profile=slim
